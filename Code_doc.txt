import pandas as pd
from pathlib import Path

# ---------------------------------------------------------
# 1. File paths
# ---------------------------------------------------------
Funnel_path = Path(
    r"\\nabdc01sdprs01\c05173_share_s_01\TOR1\Retail Fraud Strategic Analytics\...\funnel_file.csv"
)

FIQ_Data_path = Path(
    r"\\nabdc01sdprs01\c05173_share_s_01\TOR1\Retail Fraud Strategic Analytics\...\fiq_data_file.csv"
)

# ---------------------------------------------------------
# 2. Read Funnel and FIQ files
# ---------------------------------------------------------
funnel_df = pd.read_csv(
    Funnel_path,
    dtype=str,
    low_memory=False
)

fiq_df = pd.read_csv(
    FIQ_Data_path,
    dtype=str,
    low_memory=False
)

# Remove extra spaces from column names
funnel_df.columns = funnel_df.columns.str.strip()
fiq_df.columns = fiq_df.columns.str.strip()
final_df_v1.columns = final_df_v1.columns.str.strip()

# ---------------------------------------------------------
# 3. Define join columns
# ---------------------------------------------------------
# Change these based on the actual column names in the files.
funnel_fiq_join_key = "Application_ID"

# Column used to join final_df_v1 to the Funnel + FIQ dataset
final_join_key = "Application_ID"

# ---------------------------------------------------------
# 4. Avoid duplicate records before joining
# ---------------------------------------------------------
# Use this only if one application should have one FIQ record.
fiq_df = fiq_df.drop_duplicates(
    subset=[funnel_fiq_join_key],
    keep="last"
)

# ---------------------------------------------------------
# 5. Left join all FIQ columns to Funnel
# ---------------------------------------------------------
funnel_fiq_df = funnel_df.merge(
    fiq_df,                              # All FIQ columns are included
    on=funnel_fiq_join_key,
    how="left",
    suffixes=("_funnel", "_fiq"),
    validate="m:1"                       # Many funnel rows to one FIQ row
)

# ---------------------------------------------------------
# 6. Prepare final_df_v1
# ---------------------------------------------------------
# Use this if final_df_v1 should contain one row per application.
final_df_v1 = final_df_v1.drop_duplicates(
    subset=[final_join_key],
    keep="last"
)

# ---------------------------------------------------------
# 7. Use Funnel + FIQ as the main/base dataframe and
#    left join final_df_v1
# ---------------------------------------------------------
final_merged_df = funnel_fiq_df.merge(
    final_df_v1,
    on=final_join_key,
    how="left",
    suffixes=("", "_final"),
    validate="m:1"
)

# ---------------------------------------------------------
# 8. Optional cleanup
# ---------------------------------------------------------
final_merged_df.drop_duplicates(inplace=True)
final_merged_df.reset_index(drop=True, inplace=True)

print("Funnel rows:", len(funnel_df))
print("FIQ rows:", len(fiq_df))
print("Funnel + FIQ rows:", len(funnel_fiq_df))
print("Final merged rows:", len(final_merged_df))
print("Final columns:", len(final_merged_df.columns))
