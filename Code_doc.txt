import pandas as pd

# Ensure Run_Date is datetime
MBNA_blocks_hist_status["Run_Date"] = pd.to_datetime(
    MBNA_blocks_hist_status["Run_Date"],
    errors="coerce"
)

# Daily, weekly and monthly reporting fields
MBNA_blocks_hist_status["Run_Day"] = (
    MBNA_blocks_hist_status["Run_Date"].dt.normalize()
)

# Monday as the start of the week
MBNA_blocks_hist_status["Run_Week"] = (
    MBNA_blocks_hist_status["Run_Date"]
    .dt.to_period("W-SUN")
    .dt.start_time
)

# Keep year and month together so different years are not combined
MBNA_blocks_hist_status["Run_Month"] = (
    MBNA_blocks_hist_status["Run_Date"]
    .dt.to_period("M")
    .astype(str)
)

# Create conditional account-number columns
MBNA_blocks_hist_status["acct_still_blocked"] = (
    MBNA_blocks_hist_status["ACCT_NUM_TA1"]
    .where(MBNA_blocks_hist_status["susp_ind_base"].eq(1))
)

MBNA_blocks_hist_status["acct_marked_fraud"] = (
    MBNA_blocks_hist_status["ACCT_NUM_TA1"].where(
        MBNA_blocks_hist_status["susp_ind_base"].eq(0)
        & MBNA_blocks_hist_status["Cnf_fraud_ind"].eq(1)
    )
)

MBNA_blocks_hist_status["acct_unblocked"] = (
    MBNA_blocks_hist_status["ACCT_NUM_TA1"].where(
        MBNA_blocks_hist_status["susp_ind_base"].eq(0)
        & MBNA_blocks_hist_status["Cnf_fraud_ind"].eq(0)
    )
)


def create_kpi_report(df, group_column):
    return (
        df.groupby(group_column, dropna=False)
        .agg(
            num_blocked=("ACCT_NUM_TA1", "nunique"),
            num_still_blocked=("acct_still_blocked", "nunique"),
            num_marked_fraud=("acct_marked_fraud", "nunique"),
            num_unblocked=("acct_unblocked", "nunique")
        )
        .reset_index()
        .sort_values(group_column)
    )


# Daily performance
kpi_df_daily = create_kpi_report(
    MBNA_blocks_hist_status,
    "Run_Day"
)

# Weekly performance
kpi_df_weekly = create_kpi_report(
    MBNA_blocks_hist_status,
    "Run_Week"
)

# Add week-end date for reporting
kpi_df_weekly["Week_End"] = (
    kpi_df_weekly["Run_Week"] + pd.Timedelta(days=6)
)

# Reorder weekly columns
kpi_df_weekly = kpi_df_weekly[
    [
        "Run_Week",
        "Week_End",
        "num_blocked",
        "num_still_blocked",
        "num_marked_fraud",
        "num_unblocked"
    ]
]

# Monthly performance
kpi_df_monthly = create_kpi_report(
    MBNA_blocks_hist_status,
    "Run_Month"
)

print("Daily Performance")
display(kpi_df_daily)

print("Weekly Performance")
display(kpi_df_weekly)

print("Monthly Performance")
display(kpi_df_monthly)
