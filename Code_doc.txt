# Filter
filtered_df = amz_om5_final_df[
    (amz_om5_final_df["Address_Province"] == "QC") &
    (~amz_om5_final_df["recommendeddecision"].isin([366, 367, 368]))
]

# Overall monthly summary
overall_summary = (
    amz_om5_final_df
    .groupby("Application_mon")
    .agg(
        Total_Applications=("Application_mon", "size"),
        Total_Frauds=("fraud_ind", lambda x: (x == 1).sum()),
        Total_Confirmed_Frauds=("conf_fraud_ind", lambda x: (x == 1).sum()),
        Total_Suspected_Frauds=("suspected_fraud", lambda x: (x == 1).sum()),
        Total_Cur_Bal=("cur_bal_tot", "sum")
    )
    .reset_index()
)
