from ctypes import cast, POINTER
from comtypes import CLSCTX_ALL
from pycaw.pycaw import AudioUtilities, IAudioEndpointVolume
import time

def get_volume_interface():
    devices = AudioUtilities.GetSpeakers()
    interface = devices._dev.Activate(IAudioEndpointVolume._iid_, CLSCTX_ALL, None)
    return cast(interface, POINTER(IAudioEndpointVolume))

def change_volume_by_one_step(up=True):
    volume = get_volume_interface()
    current = volume.GetMasterVolumeLevelScalar()
    
    step = 0.02   # roughly one volume step
    if up:
        new_vol = min(1.0, current + step)
    else:
        new_vol = max(0.0, current - step)
    
    volume.SetMasterVolumeLevelScalar(new_vol, None)

def main():
    print("Volume toggler started. Press Ctrl+C to stop.")
    try:
        while True:
            change_volume_by_one_step(up=True)    # Volume up
            time.sleep(0.3)
            change_volume_by_one_step(up=False)   # Volume down
            time.sleep(30)                        # Wait 30 seconds (change if needed)
    except KeyboardInterrupt:
        print("\nStopped.")

if __name__ == "__main__":
    main()
