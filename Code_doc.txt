import pandas as pd

def summarize(df):
    return pd.Series({
        "Total Applications": df.shape[0],
        "Total Frauds": df["fraud_ind"].eq(1).sum(),
        "Total Confirmed Frauds": df["conf_fraud_ind"].eq(1).sum(),
        "Total Suspected Frauds": df["suspected_fraud"].eq(1).sum(),
        "Total Current Balance": df["cur_bal_tot"].sum()
    })

overall_summary = summarize(amz_om5_final_df)

filtered_df = amz_om5_final_df[
    (amz_om5_final_df["Address_Province"] == "QC") &
    (~amz_om5_final_df["recommendeddecision"].isin([366, 367, 368]))
]

filtered_summary = summarize(filtered_df)

summary = pd.concat(
    [overall_summary, filtered_summary],
    axis=1
)

summary.columns = ["Overall", "QC & Decision Not 366/367/368"]

print(summary)
