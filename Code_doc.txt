# ============================================================
# 3. SCORE ANALYSIS FUNCTION
# ============================================================

def analyse_mbna_score(
    df,
    score_name,
    score_col,
    consortium_thresholds,
    population_targets
):
    """
    Calculates:
      1. MBNA population percentage at Consortium thresholds.
      2. MBNA-equivalent thresholds for 1%, 3%, and 5%.
      3. Confirmed and suspected fraud capture rates.
    """

    # Keep one row per application and score
    score_df = (
        df
        .select(
            application_col,
            score_col,
            confirmed_fraud_col,
            suspected_fraud_col
        )
        .filter(
            F.col(application_col).isNotNull() &
            F.col(score_col).isNotNull()
        )
        .groupBy(application_col)
        .agg(
            F.max(F.col(score_col)).alias(score_col),
            F.max(F.col(confirmed_fraud_col)).alias(confirmed_fraud_col),
            F.max(F.col(suspected_fraud_col)).alias(suspected_fraud_col)
        )
        .cache()
    )

    total_apps_for_score = score_df.count()

    # --------------------------------------------------------
    # A. Results at Consortium thresholds
    # --------------------------------------------------------

    consortium_expressions = []

    for threshold in consortium_thresholds:
        consortium_expressions.extend([
            F.sum(
                F.when(F.col(score_col) > threshold, 1).otherwise(0)
            ).alias(f"applications_{threshold}"),

            F.sum(
                F.when(
                    (F.col(score_col) > threshold) &
                    (F.col(confirmed_fraud_col) == 1),
                    1
                ).otherwise(0)
            ).alias(f"confirmed_{threshold}"),

            F.sum(
                F.when(
                    (F.col(score_col) > threshold) &
                    (F.col(suspected_fraud_col) == 1),
                    1
                ).otherwise(0)
            ).alias(f"suspected_{threshold}")
        ])

    consortium_counts = (
        score_df
        .agg(*consortium_expressions)
        .first()
        .asDict()
    )

    consortium_results = []

    for population_target, threshold in zip(
        population_targets,
        consortium_thresholds
    ):
        application_count = consortium_counts[f"applications_{threshold}"]
        confirmed_count = consortium_counts[f"confirmed_{threshold}"]
        suspected_count = consortium_counts[f"suspected_{threshold}"]

        consortium_results.append({
            "Score Type": score_name,
            "% Population": population_target,
            "Consortium Score": threshold,
            "MBNA Population Rate":
                application_count / total_apps_for_score
                if total_apps_for_score else 0,
            "MBNA Application Count": application_count,
            "Confirmed Fraud Captured": confirmed_count,
            "Confirmed Fraud Capture Rate":
                confirmed_count / total_confirmed_fraud
                if total_confirmed_fraud else 0,
            "Suspected Fraud Captured": suspected_count,
            "Suspected Fraud Capture Rate":
                suspected_count / total_suspected_fraud
                if total_suspected_fraud else 0
        })

    # --------------------------------------------------------
    # B. MBNA-equivalent score thresholds
    # --------------------------------------------------------
    # To select approximately the highest 1%, 3%, and 5%,
    # calculate the 99th, 97th, and 95th percentiles.

    percentiles = [1 - target for target in population_targets]

    percentile_array = ", ".join(str(x) for x in percentiles)

    mbna_thresholds = (
        score_df
        .agg(
            F.expr(
                f"percentile_approx("
                f"{score_col}, "
                f"array({percentile_array}), "
                f"10000)"
            ).alias("thresholds")
        )
        .first()["thresholds"]
    )

    mbna_expressions = []

    for position, threshold in enumerate(mbna_thresholds):
        mbna_expressions.extend([
            F.sum(
                F.when(F.col(score_col) > threshold, 1).otherwise(0)
            ).alias(f"applications_{position}"),

            F.sum(
                F.when(
                    (F.col(score_col) > threshold) &
                    (F.col(confirmed_fraud_col) == 1),
                    1
                ).otherwise(0)
            ).alias(f"confirmed_{position}"),

            F.sum(
                F.when(
                    (F.col(score_col) > threshold) &
                    (F.col(suspected_fraud_col) == 1),
                    1
                ).otherwise(0)
            ).alias(f"suspected_{position}")
        ])

    mbna_counts = (
        score_df
        .agg(*mbna_expressions)
        .first()
        .asDict()
    )

    equivalent_results = []

    for position, (
        population_target,
        consortium_threshold,
        mbna_threshold
    ) in enumerate(
        zip(
            population_targets,
            consortium_thresholds,
            mbna_thresholds
        )
    ):
        application_count = mbna_counts[f"applications_{position}"]
        confirmed_count = mbna_counts[f"confirmed_{position}"]
        suspected_count = mbna_counts[f"suspected_{position}"]

        equivalent_results.append({
            "Score Type": score_name,
            "% Population": population_target,
            "Consortium Score": consortium_threshold,
            "MBNA Equivalent Score": mbna_threshold,
            "Actual MBNA Population Rate":
                application_count / total_apps_for_score
                if total_apps_for_score else 0,
            "MBNA Application Count": application_count,
            "Confirmed Fraud Captured": confirmed_count,
            "Confirmed Fraud Capture Rate":
                confirmed_count / total_confirmed_fraud
                if total_confirmed_fraud else 0,
            "Suspected Fraud Captured": suspected_count,
            "Suspected Fraud Capture Rate":
                suspected_count / total_suspected_fraud
                if total_suspected_fraud else 0
        })

    score_df.unpersist()

    return consortium_results, equivalent_results


# ============================================================
# 4. RUN ANALYSIS
# ============================================================

all_consortium_results = []
all_equivalent_results = []

for score_name, config in score_config.items():

    consortium_result, equivalent_result = analyse_mbna_score(
        df=mbna_df,
        score_name=score_name,
        score_col=config["column"],
        consortium_thresholds=config["consortium_thresholds"],
        population_targets=config["population_targets"]
    )

    all_consortium_results.extend(consortium_result)
    all_equivalent_results.extend(equivalent_result)


consortium_results_pdf = pd.DataFrame(all_consortium_results)
equivalent_results_pdf = pd.DataFrame(all_equivalent_results)

excel_tables = {}

for score_name in score_config.keys():

    consortium_part = (
        consortium_results_pdf[
            consortium_results_pdf["Score Type"] == score_name
        ]
        .copy()
        .sort_values("% Population")
    )

    equivalent_part = (
        equivalent_results_pdf[
            equivalent_results_pdf["Score Type"] == score_name
        ]
        .copy()
        .sort_values("% Population")
    )

    # Top table: Consortium thresholds and resulting MBNA population
    population_table = pd.DataFrame({
        "Score": consortium_part["Consortium Score"].values,
        "Consortium": consortium_part["% Population"].values,
        "MBNA": consortium_part["MBNA Population Rate"].values
    })

    # Bottom table: score corresponding to 1%, 3%, and 5%
    equivalent_score_table = pd.DataFrame({
        "% Population": equivalent_part["% Population"].values,
        "Consortium": equivalent_part["Consortium Score"].values,
        "MBNA": equivalent_part["MBNA Equivalent Score"].values
    })

    # Fraud capture using Consortium thresholds
    fraud_capture_table = pd.DataFrame({
        "Score": consortium_part["Consortium Score"].values,
        "MBNA Population": consortium_part["MBNA Population Rate"].values,
        "Confirmed Fraud Capture":
            consortium_part["Confirmed Fraud Capture Rate"].values,
        "Suspected Fraud Capture":
            consortium_part["Suspected Fraud Capture Rate"].values
    })

    # Fraud capture using MBNA-equivalent thresholds
    equivalent_fraud_table = pd.DataFrame({
        "% Population": equivalent_part["% Population"].values,
        "MBNA Score": equivalent_part["MBNA Equivalent Score"].values,
        "Actual MBNA Population":
            equivalent_part["Actual MBNA Population Rate"].values,
        "Confirmed Fraud Capture":
            equivalent_part["Confirmed Fraud Capture Rate"].values,
        "Suspected Fraud Capture":
            equivalent_part["Suspected Fraud Capture Rate"].values
    })

    excel_tables[score_name] = {
        "population": population_table,
        "equivalent": equivalent_score_table,
        "fraud": fraud_capture_table,
        "equivalent_fraud": equivalent_fraud_table
    }
