def path_exists(path: str) -> bool:
    """
    Check whether a Databricks/DBFS/Volume path exists.
    """
    try:
        dbutils.fs.ls(path)
        return True
    except Exception:
        return False

def normalize_column_names(df: DataFrame) -> DataFrame:
    """
    Standardize column names for CSV output and future reads.

    Replaces spaces and selected punctuation with underscores.
    """
    output_df = df

    for old_name in df.columns:
        new_name = (
            old_name.strip()
            .replace(" ", "_")
            .replace("-", "_")
            .replace("/", "_")
            .replace(".", "_")
            .replace("(", "")
            .replace(")", "")
        )

        while "__" in new_name:
            new_name = new_name.replace("__", "_")

        if old_name != new_name:
            output_df = output_df.withColumnRenamed(old_name, new_name)

    return output_df

def align_dataframes(
    first_df: DataFrame,
    second_df: DataFrame
) -> tuple[DataFrame, DataFrame]:
    """
    Add missing columns to each DataFrame so unionByName can be used safely.
    """
    first_columns = set(first_df.columns)
    second_columns = set(second_df.columns)

    for column_name in second_columns - first_columns:
        data_type = second_df.schema[column_name].dataType
        first_df = first_df.withColumn(
            column_name,
            F.lit(None).cast(data_type)
        )

    for column_name in first_columns - second_columns:
        data_type = first_df.schema[column_name].dataType
        second_df = second_df.withColumn(
            column_name,
            F.lit(None).cast(data_type)
        )

    ordered_columns = sorted(set(first_df.columns).union(second_df.columns))

    return (
        first_df.select(*ordered_columns),
        second_df.select(*ordered_columns)
    )

def read_csv_if_exists(
    path: str,
    schema: StructType | None = None
) -> DataFrame | None:
    """
    Read a CSV folder if it exists.

    Spark writes CSV output as a directory containing part files.
    """
    if not path_exists(path):
        return None

    reader = (
        spark.read
        .option("header", True)
        .option("inferSchema", schema is None)
    )

    if schema is not None:
        reader = reader.schema(schema)

    return reader.csv(path)

def get_new_records(
    current_df: DataFrame,
    history_df: DataFrame | None,
    key_columns: list[str]
) -> DataFrame:
    """
    Return only current records that do not already exist in history.

    Uses null-safe string keys and a Spark left-anti join.
    """
    current_df = current_df.dropDuplicates(key_columns)

    if history_df is None:
        return current_df

    available_keys = [
        column_name
        for column_name in key_columns
        if column_name in current_df.columns
        and column_name in history_df.columns
    ]

    if not available_keys:
        raise ValueError(
            "No valid history key columns exist in both current and "
            "historical DataFrames."
        )

    current_keys = current_df.select(
        *[
            F.coalesce(
                F.col(column_name).cast("string"),
                F.lit("__NULL__")
            ).alias(column_name)
            for column_name in available_keys
        ]
    ).dropDuplicates()

    history_keys = history_df.select(
        *[
            F.coalesce(
                F.col(column_name).cast("string"),
                F.lit("__NULL__")
            ).alias(column_name)
            for column_name in available_keys
        ]
    ).dropDuplicates()

    current_with_keys = current_df

    for column_name in available_keys:
        current_with_keys = current_with_keys.withColumn(
            f"__key_{column_name}",
            F.coalesce(
                F.col(column_name).cast("string"),
                F.lit("__NULL__")
            )
        )

    history_keys_renamed = history_keys.select(
        *[
            F.col(column_name).alias(f"__key_{column_name}")
            for column_name in available_keys
        ]
    )

    join_columns = [
        f"__key_{column_name}"
        for column_name in available_keys
    ]

    new_records = current_with_keys.join(
        history_keys_renamed,
        on=join_columns,
        how="left_anti"
    )

    return new_records.drop(*join_columns)

def write_csv(
    df: DataFrame,
    path: str,
    mode: str = "overwrite",
    single_file: bool = False
) -> None:
    """
    Write a Spark DataFrame as CSV.

    single_file=True uses coalesce(1), which is appropriate only for
    reasonably small operational reports.
    """
    output_df = df.coalesce(1) if single_file else df

    (
        output_df.write
        .format("csv")
        .mode(mode)
        .option("header", True)
        .option("emptyValue", "")
        .option("nullValue", "")
        .option("timestampFormat", "yyyy-MM-dd HH:mm:ss")
        .option("dateFormat", "yyyy-MM-dd")
        .save(path)
    )

def create_history_backup(
    history_path: str,
    backup_base_path: str,
    run_timestamp: datetime
) -> None:
    """
    Create a dated copy of the existing historical CSV folder.
    """
    if not path_exists(history_path):
        print("No historical file exists yet. Backup skipped.")
        return

    timestamp_string = run_timestamp.strftime("%Y%m%d_%H%M%S")
    backup_path = f"{backup_base_path}/history_{timestamp_string}"

    dbutils.fs.cp(
        history_path,
        backup_path,
        recurse=True
    )

    print(f"History backup created: {backup_path}")

BUSINESS_CALENDAR_TABLE = (
    "hive_metastore.<schema>.<business_calendar_table>"
)

def is_processing_day(process_date: date) -> bool:
    """
    Return True when the process should run.

    When no calendar table is configured, weekends are excluded and
    weekdays are processed.
    """
    if process_date.weekday() >= 5:
        return False

    if not USE_BUSINESS_DAY_CHECK:
        return True

    calendar_result = (
        spark.table(BUSINESS_CALENDAR_TABLE)
        .filter(F.col("calendar_date") == F.lit(process_date))
        .select(
            F.coalesce(
                F.col("is_business_day"),
                F.lit(False)
            ).alias("is_business_day")
        )
        .limit(1)
        .collect()
    )

    if not calendar_result:
        raise ValueError(
            f"No calendar record found for {process_date}."
        )

    return bool(calendar_result[0]["is_business_day"])

def rename_single_csv_part(
    source_directory: str,
    final_file_name: str
) -> str:
    """
    Rename a single Spark CSV part file inside its output directory.

    The final file remains within the same output directory.
    """
    files = dbutils.fs.ls(source_directory)

    part_files = [
        file_info.path
        for file_info in files
        if file_info.name.startswith("part-")
        and file_info.name.endswith(".csv")
    ]

    if len(part_files) != 1:
        raise ValueError(
            f"Expected one CSV part file in {source_directory}, "
            f"but found {len(part_files)}."
        )

    destination_path = (
        f"{source_directory.rstrip('/')}/{final_file_name}"
    )

    dbutils.fs.mv(
        part_files[0],
        destination_path
    )

    return destination_path
